from web3 import Web3 import time # Подключение wss = "wss://mainnet.infura.io/ws/v3/YOUR_INFURA_KEY" web3 = Web3(Web3.WebsocketProvider(wss)) token_address = Web3.toChecksumAddress("0xdAC17F958D2ee523a2206206994597C13D831ec7") abi = [ { "anonymous": False, "inputs": [ {"indexed": True, "name": "owner", "type": "address"}, {"indexed": True, "name": "spender", "type": "address"}, {"indexed": False, "name": "value", "type": "uint256"} ], "name": "Approval", "type": "event" } ] contract = web3.eth.contract(address=token_address, abi=abi) event_filter = contract.events.Approval.createFilter(fromBlock='latest') # Статистика TOTAL_AVAILABLE = 100 claimed = 0 print("🎁 Стартуем...") while True: for event in event_filter.get_new_entries(): owner = event['args']['owner'] spender = event['args']['spender'] value = event['args']['value'] / 10**6 claimed += 1 remaining = TOTAL_AVAILABLE - claimed print("\n✅ Новое разрешение:") print(f"- От: {owner}") print(f"- Кому: {spender}") print(f"- Сумма: {value} USDT") print(f"- Забрано: {claimed} / {TOTAL_AVAILABLE}") print(f"- Осталось: {remaining}") time.sleep(2)
